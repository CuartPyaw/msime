cmake_minimum_required(VERSION 3.25)
project(MSIMEWindows7Compatibility LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Win7 Rust 兼容对象需要 MSVC 延迟加载器")
endif()
include(CTest)
add_library(windows7-delay-hook OBJECT Windows7DelayLoad.cpp)
target_compile_features(windows7-delay-hook PRIVATE cxx_std_17)
target_compile_options(windows7-delay-hook PRIVATE /W4 /WX /utf-8)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/windows7-delay-object-$<CONFIG>.txt"
  CONTENT "$<TARGET_OBJECTS:windows7-delay-hook>")
add_executable(windows7-delay-load ../tests/runtime/windows7_delay_load.cpp $<TARGET_OBJECTS:windows7-delay-hook>)
target_compile_features(windows7-delay-load PRIVATE cxx_std_17)
target_compile_options(windows7-delay-load PRIVATE /W4 /WX /utf-8 /UNDEBUG)
target_link_libraries(windows7-delay-load PRIVATE ole32 oleaut32)
add_executable(windows7-core-load ../tests/runtime/windows7_core_load.cpp)
target_compile_features(windows7-core-load PRIVATE cxx_std_17)
target_compile_options(windows7-core-load PRIVATE /W4 /WX /utf-8 /UNDEBUG)
target_link_libraries(windows7-core-load PRIVATE ole32)
add_test(NAME windows7-delay-load COMMAND windows7-delay-load)
set_tests_properties(windows7-delay-load PROPERTIES TIMEOUT 30)
