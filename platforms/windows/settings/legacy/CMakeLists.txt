cmake_minimum_required(VERSION 3.25)
project(MSIMEWindowsLegacySettings LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "兼容设置窗口需要 Windows 原生工具链")
endif()
include(../../Edition.cmake)
include(CTest)
find_package(nlohmann_json 3.11 REQUIRED)
set(MSIME_HOST_LIBRARY "" CACHE FILEPATH "同源 Win7 核心 DLL 的导入库")
if(NOT IS_ABSOLUTE "${MSIME_HOST_LIBRARY}" OR NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "MSIME_HOST_LIBRARY 必须是同源核心的绝对导入库路径")
endif()
add_library(msime-legacy-settings-window STATIC SettingsWindow.cpp)
target_include_directories(msime-legacy-settings-window PUBLIC ../../../../crates/host-api/include)
target_compile_features(msime-legacy-settings-window PUBLIC cxx_std_17)
target_compile_definitions(msime-legacy-settings-window PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(msime-legacy-settings-window PUBLIC nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}" comctl32 ole32 shell32 advapi32 user32 gdi32)
if(MSVC)
  target_compile_options(msime-legacy-settings-window PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(msime-legacy-settings-window PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(msime-client-settings-legacy WIN32 main.cpp Settings.rc)
target_link_libraries(msime-client-settings-legacy PRIVATE msime-legacy-settings-window)
if(MSVC)
  # 清单已由 Settings.rc 嵌入，不能再让链接器生成同编号资源。
  target_link_options(msime-client-settings-legacy PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(msime-client-settings-legacy PRIVATE -municode)
endif()
if(BUILD_TESTING)
  add_executable(windows-legacy-settings-test ../../tests/ui/legacy_settings.cpp Settings.rc)
  target_link_libraries(windows-legacy-settings-test PRIVATE msime-legacy-settings-window)
  if(MSVC)
    target_compile_options(windows-legacy-settings-test PRIVATE /W4 /WX /utf-8 /UNDEBUG)
    target_link_options(windows-legacy-settings-test PRIVATE /MANIFEST:NO)
  else()
    target_compile_options(windows-legacy-settings-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME windows-legacy-settings COMMAND windows-legacy-settings-test)
  set_tests_properties(windows-legacy-settings PROPERTIES TIMEOUT 45)
endif()
